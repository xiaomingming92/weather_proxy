private static final int[] TABLE_CONDITION_CHINA2ACCU = new int[]{
    1, 6, 8, 18, 15, 51, 29, 14, 13, 18, 15, 22, 22, 23, 21, 19, 
    22, 22, 11, 26, 52, 13, 15, 15, 15, 15, 19, 22, 22, 53, 52, 52,

    33, 38, 8, 18, 15, 51, 29, 14, 13, 18, 15, 22, 22, 23, 21, 19, 
    22, 22, 11, 26, 52, 13, 15, 15, 15, 15, 19, 22, 22, 53, 52, 52,

### 5.2 转换方法
```java
private String China2AccuCondition(String condition) {
    int code = Integer.parseInt(condition);
    if (code >= 0 && code < TABLE_CONDITION_CHINA2ACCU.length) {
        return Integer.valueOf(TABLE_CONDITION_CHINA2ACCU[code]).toString();
    }
    return "0";
}

private String China2AccuConditionWithNight(String condition) {
    int code = Integer.parseInt(condition);
    Calendar calendar = Calendar.getInstance(TimeZone.getTimeZone("Asia/Hong_Kong"));
    int hour = calendar.getTime().getHours();
    
    if (code >= 0 && code < TABLE_CONDITION_CHINA2ACCU.length) {
        if (hour > 6 && hour < 18) {
            // 白天
            return Integer.valueOf(TABLE_CONDITION_CHINA2ACCU[code]).toString();
        } else {
            // 夜间
            return Integer.valueOf(TABLE_NIGHT_CONDITION_CHINA2ACCU[code]).toString();
        }
    }
    return "nop";
}
```
